
(function () {
  "use strict";

  // L'éditeur ne fonctionne que sur la page admin.
  if (document.body.dataset.mapMode !== "admin") return;

  window.D8InitZoneEditor = function (map) {
    if (!map || !window.L) {
      console.error("Carte Leaflet indisponible.");
      return;
    }

    if (document.getElementById("d8-draw-tools")) return;

    const drawing = {
      enabled: false,
      points: [],
      polygon: null,
      markers: [],
      color: "#bb65ff"
    };

    const drawTools = document.createElement("div");
    drawTools.id = "d8-draw-tools";

   
drawTools.innerHTML = `
  <div class="d8-prestige-editor-head">
    <span class="d8-prestige-editor-crown">♛</span>
    <div class="d8-draw-title">ÉDITION DES ZONES</div>
    <div class="d8-prestige-editor-subtitle">
      GESTION DES QUARTIERS DE LOS SANTOS
    </div>
  </div>

  <div class="d8-prestige-editor-grid">

    <div class="d8-prestige-editor-section">
      <div class="d8-prestige-editor-label">
        ✦ GESTION DES QUARTIERS
      </div>

      <button id="d8-zone-new" type="button">
        ➕ Nouveau quartier
      </button>

      <button id="d8-zone-select" type="button">
        📍 Sélectionner un quartier
      </button>

      <button id="d8-draw-toggle" type="button">
        ✏️ Activer le dessin
      </button>

      <button id="d8-draw-undo" type="button">
        ↶ Annuler un point
      </button>

      <button id="d8-draw-copy" type="button">
        📋 Copier les coordonnées
      </button>

      <button id="d8-draw-clear" type="button">
        ✕ Effacer le tracé
      </button>

      <div id="d8-draw-count">0 point</div>
    </div>

    <div class="d8-prestige-editor-section">
      <div class="d8-prestige-editor-label">
        ✦ INFORMATIONS DU QUARTIER
      </div>

      <div id="d8-zone-form" hidden>
        <label for="d8-zone-name">Nom du quartier</label>
        <input id="d8-zone-name" type="text"
          maxlength="100"
          placeholder="Ex : Vinewood Hills">

        <label for="d8-zone-type">Type de quartier</label>
        <input id="d8-zone-type" type="text"
          maxlength="100"
          placeholder="Ex : Résidentiel de luxe">

        <label for="d8-zone-category">Catégorie</label>
        <select id="d8-zone-category">
          <option value="lux">💜 Luxe</option>
          <option value="res">💙 Résidentiel</option>
          <option value="rural">🩵 Rural</option>
          <option value="ind">🩷 Industriel</option>
          <option value="excluded">🚫 Zone exclue</option>
        </select>

        <label for="d8-zone-description">Description</label>
        <textarea id="d8-zone-description"
          rows="3"
          placeholder="Description du quartier..."></textarea>
      </div>

      <div class="d8-prestige-editor-empty">
        Sélectionnez ou créez un quartier
        pour modifier ses informations.
      </div>
    </div>
  </div>

  <div class="d8-prestige-editor-actions">
    <button id="d8-zone-save" type="button">
      💾 Enregistrer les modifications
    </button>

    <button id="d8-zone-delete" type="button">
      🗑️ Supprimer le quartier
    </button>
  </div>
`;


    document.body.appendChild(drawTools);
const zoneNew = document.getElementById("d8-zone-new");
const zoneSelect = document.getElementById("d8-zone-select");
let selectingZone = false;
let selectedZoneId = null;
let creatingNewZone = false;
    const drawToggle = document.getElementById("d8-draw-toggle");
    const drawCount = document.getElementById("d8-draw-count");
const zoneSave = document.getElementById("d8-zone-save");
   const zoneDelete = document.getElementById("d8-zone-delete");
const zoneForm = document.getElementById("d8-zone-form");
const zoneName = document.getElementById("d8-zone-name");
const zoneType = document.getElementById("d8-zone-type");
const zoneCategory = document.getElementById("d8-zone-category");
const zoneDescription = document.getElementById("d8-zone-description");

function refreshDrawing() {
      if (drawing.polygon) {
        map.removeLayer(drawing.polygon);
        drawing.polygon = null;
      }

      drawing.markers.forEach(marker => map.removeLayer(marker));
      drawing.markers = [];

      if (drawing.points.length >= 2) {
        drawing.polygon = L.polygon(drawing.points, {
          color: drawing.color,
          weight: 4,
          fillColor: drawing.color,
          fillOpacity: 0.15,
          interactive: false
        }).addTo(map);
      }

      
drawing.points.forEach((point, index) => {
  const marker = L.circleMarker(point, {
    radius: 6,
    color: "#ffffff",
    fillColor: drawing.color,
    fillOpacity: 1,
    weight: 2,
    interactive: true,
    bubblingMouseEvents: false
  }).addTo(map);

  marker.on("mousedown", event => {
    if (!selectedZoneId || drawing.enabled) return;

    L.DomEvent.stopPropagation(event);

    map.dragging.disable();

    const movePoint = moveEvent => {
      drawing.points[index] = [
        Math.round(moveEvent.latlng.lat),
        Math.round(moveEvent.latlng.lng)
      ];

      marker.setLatLng(drawing.points[index]);

      if (drawing.polygon) {
        drawing.polygon.setLatLngs(drawing.points);
      }
    };

    const stopMoving = () => {
      map.off("mousemove", movePoint);
      map.off("mouseup", stopMoving);
      map.dragging.enable();
    };

    map.on("mousemove", movePoint);
    map.on("mouseup", stopMoving);
  });

  drawing.markers.push(marker);
});


      drawCount.textContent =
        drawing.points.length +
        (drawing.points.length > 1 ? " points" : " point");
    }

zoneNew.addEventListener("click", () => {
  if (!confirm("Commencer la création d'un nouveau quartier ?")) {
    return;
  }

  // Quitter le mode sélection
  selectingZone = false;
  selectedZoneId = null;
creatingNewZone = true;
  document.body.classList.remove("d8-selecting-zone");

  zoneSelect.textContent = "📍 Sélectionner un quartier";

  // Effacer uniquement le tracé de travail
  drawing.points = [];
  drawing.color = "#bb65ff";
  refreshDrawing();

  // Préparer les informations
  zoneName.value = "";
  zoneType.value = "";
  zoneCategory.value = "res";
  zoneDescription.value = "";
  zoneForm.hidden = false;

  // Activer le dessin
  if (!drawing.enabled) {
    drawToggle.click();
  }

  alert(
    "Nouveau quartier : clique sur la carte pour dessiner ses contours."
  );
});

zoneSelect.addEventListener("click", () => {
  selectingZone = !selectingZone;

  if (selectingZone && drawing.enabled) {
    drawToggle.click();
  }

  zoneSelect.textContent = selectingZone
    ? "✓ Cliquez sur un quartier"
    : "📍 Sélectionner un quartier";

  map.getContainer().style.cursor =
    selectingZone ? "pointer" : "";
});

map.on("click", event => {
  if (!selectingZone) return;

  const zones = window.D8Zones?.zones;

  if (!zones) {
    alert("Les quartiers ne sont pas encore chargés.");
    return;
  }

  
const pointIsInsidePolygon = (latlng, points) => {
  const x = latlng.lng;
  const y = latlng.lat;
  let inside = false;

  for (
    let i = 0, j = points.length - 1;
    i < points.length;
    j = i++
  ) {
    const yi = points[i][0];
    const xi = points[i][1];
    const yj = points[j][0];
    const xj = points[j][1];

    const intersects =
      (yi > y) !== (yj > y) &&
      x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;

    if (intersects) {
      inside = !inside;
    }
  }

  return inside;
};

const clickedZone = Object.entries(zones)
  .reverse()
  .find(([id, zone]) => {
    return pointIsInsidePolygon(
      event.latlng,
      zone.points
    );
  });


  if (!clickedZone) {
    alert("Aucun quartier trouvé à cet endroit.");
    return;
  }

  const [id, zone] = clickedZone;

  selectedZoneId = id;
creatingNewZone = false;

  drawing.points = zone.points.map(point => [...point]);

zoneName.value = zone.name || "";
zoneType.value = zone.type || "";
zoneCategory.value = zone.category || "res";
zoneDescription.value = zone.description || "";

zoneForm.hidden = false;

  const colors = window.D8Zones.colors;
  drawing.color = colors[zone.category] || "#bb65ff";

  refreshDrawing();

  selectingZone = false;
  zoneSelect.textContent = "📍 " + zone.name;
  map.getContainer().style.cursor = "";

  alert(
    "Quartier sélectionné : " + zone.name +
    "\nSes points sont maintenant affichés."
  );
});

    drawToggle.addEventListener("click", () => {
      drawing.enabled = !drawing.enabled;

      drawToggle.textContent = drawing.enabled
        ? "✓ Terminer le dessin"
        : "✏️ Activer le dessin";

      map.getContainer().style.cursor =
        drawing.enabled ? "crosshair" : "";

      drawTools.classList.toggle("drawing-active", drawing.enabled);
    });

    map.on("click", event => {
      if (!drawing.enabled) return;

      const lat = Math.round(event.latlng.lat);
      const lng = Math.round(event.latlng.lng);

      drawing.points.push([lat, lng]);
      refreshDrawing();
    });

    document.getElementById("d8-draw-undo")
      .addEventListener("click", () => {
        drawing.points.pop();
        refreshDrawing();
      });

    document.getElementById("d8-draw-clear")
      .addEventListener("click", () => {
        drawing.points = [];
        refreshDrawing();
      });

    document.getElementById("d8-draw-copy")
      .addEventListener("click", async () => {
        if (drawing.points.length < 3) {
          alert("Place au moins 3 points sur la carte.");
          return;
        }

        const coordinates =
          "points: " + JSON.stringify(drawing.points, null, 2);

        try {
          await navigator.clipboard.writeText(coordinates);
          alert("Coordonnées copiées !");
        } catch {
          window.prompt(
            "Copie ces coordonnées avec Ctrl + C :",
            coordinates
          );
        }
      });

    // =====================================
    // SAUVEGARDE SECURISEE DES QUARTIERS
    // =====================================

    let savingZone = false;

zoneDelete.addEventListener("click", async () => {
  // Empêcher la suppression pendant un enregistrement
  if (savingZone || zoneDelete.disabled) return;

  // Vérifier qu'un quartier existant est sélectionné
  if (!selectedZoneId || creatingNewZone) {
    alert("Sélectionne d'abord un quartier existant.");
    return;
  }

  const zoneId = selectedZoneId;
  const zone = window.D8Zones?.zones?.[zoneId];

  if (!zone) {
    alert("Quartier introuvable. Actualise la carte.");
    return;
  }

  // Première confirmation
  if (!confirm(
    "Voulez-vous supprimer le quartier « " +
    zone.name +
    " » ?"
  )) return;

  // Deuxième confirmation : saisir le nom exact
  const confirmation = prompt(
    "Suppression définitive.\n\n" +
    "Pour confirmer, recopie exactement le nom du quartier :\n" +
    zone.name
  );

  if (confirmation !== zone.name) {
    alert("Suppression annulée : nom incorrect.");
    return;
  }

  zoneDelete.disabled = true;
  zoneDelete.textContent = "⏳ Suppression...";

  try {
    if (typeof window.D8Cloud?.deleteZone !== "function") {
      throw new Error("Fonction de suppression indisponible.");
    }

    await window.D8Cloud.deleteZone(zoneId);

    // Réactualiser les quartiers depuis Supabase
    await window.D8Zones.refresh();

    // Réinitialiser l'éditeur
    selectedZoneId = null;
    creatingNewZone = false;
    selectingZone = false;

    document.body.classList.remove("d8-selecting-zone");

    zoneSelect.textContent = "📍 Sélectionner un quartier";

    drawing.points = [];
    drawing.color = "#bb65ff";
    refreshDrawing();

    zoneForm.hidden = true;

    alert("✅ Quartier supprimé de Supabase !");

  } catch (error) {
    console.error("Suppression du quartier :", error);
    alert(
      "❌ " + error.message +
      "\nActualise la carte pour vérifier l'état du quartier."
    );

  } finally {
    zoneDelete.disabled = false;
    zoneDelete.textContent = "🗑️ Supprimer le quartier";
  }
});

    zoneSave.addEventListener("click", async () => {
      if (savingZone) return;

      
if (!selectedZoneId && !creatingNewZone) {
  alert("Sélectionne un quartier ou crée un nouveau quartier.");
  return;
}

      if (drawing.enabled || selectingZone) {
        alert("Termine le dessin ou la sélection avant d'enregistrer.");
        return;
      }

     
const zone = creatingNewZone
  ? {
      name: zoneName.value.trim(),
      type: zoneType.value.trim(),
      category: zoneCategory.value,
      description: zoneDescription.value.trim(),
      interiors: []
    }
  : window.D8Zones?.zones?.[selectedZoneId];

if (!zone) {
  alert("Quartier introuvable. Actualise la carte.");
  return;
}


if (!zoneName.value.trim()) {
  alert("Le nom du quartier est obligatoire.");
  zoneName.focus();
  return;
}

      const points = drawing.points.map(point => [...point]);

      if (
        points.length < 3 ||
        points.length > 5000 ||
        !points.every(point =>
          Array.isArray(point) &&
          point.length === 2 &&
          point.every(value =>
            Number.isFinite(value) &&
            value >= 0 &&
            value <= 8192
          )
        )
      ) {
        alert("Les coordonnées du quartier sont invalides.");
        return;
      }

      if (!confirm(
        "Enregistrer les modifications de « " +
        zone.name +
        " » dans Supabase ?"
      )) return;

      savingZone = true;
      zoneSave.disabled = true;
      zoneSave.textContent = "⏳ Enregistrement...";

      try {
        if (typeof window.D8Cloud?.saveZone !== "function") {
          throw new Error("Fonction de sauvegarde indisponible.");
        }

       
const zoneId = creatingNewZone
  ? "zone_" + crypto.randomUUID().replace(/-/g, "")
  : selectedZoneId;

await window.D8Cloud.saveZone({
  id: zoneId,
  name: zoneName.value.trim(),
  type: zoneType.value.trim(),
  category: zoneCategory.value,
  description: zoneDescription.value.trim(),
  interiors: zone.interiors || [],
  points
});


        await window.D8Zones.refresh();

selectedZoneId = zoneId;
creatingNewZone = false;

if (drawing.enabled) {
  drawToggle.click();
}

zoneSelect.textContent = "📍 " + zoneName.value.trim();

        // Recharger le tracé depuis les données enregistrées
        const savedZone =
          window.D8Zones.zones[selectedZoneId];

        if (savedZone) {
          drawing.points = savedZone.points.map(
            point => [...point]
          );
          refreshDrawing();
        }

        alert("✅ Quartier enregistré dans Supabase !");

      } catch (error) {
        console.error("Sauvegarde du quartier :", error);
        alert("❌ " + error.message);

      } finally {
        savingZone = false;
        zoneSave.disabled = false;
        zoneSave.textContent =
          "💾 Enregistrer les modifications";
      }
    });

  };
})();
