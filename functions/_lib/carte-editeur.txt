
(function () {
  "use strict";

  // L'éditeur ne fonctionne que sur la page admin.
  if (document.body.dataset.mapMode !== "admin") return;

  window.D8InitZoneEditor = function (map) {
    if (!map || !window.L) {
      console.error("Carte Leaflet indisponible.");
      return;
    }

    if (document.getElementById("d8-draw-tools")) return;

    const drawing = {
      enabled: false,
      points: [],
      polygon: null,
      markers: [],
      color: "#bb65ff"
    };

    const drawTools = document.createElement("div");
    drawTools.id = "d8-draw-tools";

    drawTools.innerHTML = `
      <div class="d8-draw-title">✦ ÉDITION DES ZONES</div>
<button id="d8-zone-select" type="button">📍 Sélectionner un quartier</button>
      <button id="d8-draw-toggle" type="button">✏️ Activer le dessin</button>
      <button id="d8-draw-undo" type="button">↶ Annuler un point</button>
      <button id="d8-draw-copy" type="button">📋 Copier les coordonnées</button>
      <button id="d8-draw-clear" type="button">✕ Effacer le tracé</button>
      <div id="d8-draw-count">0 point</div>
    `;

    document.body.appendChild(drawTools);
const zoneSelect = document.getElementById("d8-zone-select");
let selectingZone = false;
let selectedZoneId = null;
    const drawToggle = document.getElementById("d8-draw-toggle");
    const drawCount = document.getElementById("d8-draw-count");

    function refreshDrawing() {
      if (drawing.polygon) {
        map.removeLayer(drawing.polygon);
        drawing.polygon = null;
      }

      drawing.markers.forEach(marker => map.removeLayer(marker));
      drawing.markers = [];

      if (drawing.points.length >= 2) {
        drawing.polygon = L.polygon(drawing.points, {
          color: drawing.color,
          weight: 4,
          fillColor: drawing.color,
          fillOpacity: 0.15,
          interactive: false
        }).addTo(map);
      }

      
drawing.points.forEach((point, index) => {
  const marker = L.circleMarker(point, {
    radius: 6,
    color: "#ffffff",
    fillColor: drawing.color,
    fillOpacity: 1,
    weight: 2,
    interactive: true,
    bubblingMouseEvents: false
  }).addTo(map);

  marker.on("mousedown", event => {
    if (!selectedZoneId || drawing.enabled) return;

    L.DomEvent.stopPropagation(event);

    map.dragging.disable();

    const movePoint = moveEvent => {
      drawing.points[index] = [
        Math.round(moveEvent.latlng.lat),
        Math.round(moveEvent.latlng.lng)
      ];

      marker.setLatLng(drawing.points[index]);

      if (drawing.polygon) {
        drawing.polygon.setLatLngs(drawing.points);
      }
    };

    const stopMoving = () => {
      map.off("mousemove", movePoint);
      map.off("mouseup", stopMoving);
      map.dragging.enable();
    };

    map.on("mousemove", movePoint);
    map.on("mouseup", stopMoving);
  });

  drawing.markers.push(marker);
});


      drawCount.textContent =
        drawing.points.length +
        (drawing.points.length > 1 ? " points" : " point");
    }

zoneSelect.addEventListener("click", () => {
  selectingZone = !selectingZone;

  if (selectingZone && drawing.enabled) {
    drawToggle.click();
  }

  zoneSelect.textContent = selectingZone
    ? "✓ Cliquez sur un quartier"
    : "📍 Sélectionner un quartier";

  map.getContainer().style.cursor =
    selectingZone ? "pointer" : "";
});

map.on("click", event => {
  if (!selectingZone) return;

  const zones = window.D8Zones?.zones;

  if (!zones) {
    alert("Les quartiers ne sont pas encore chargés.");
    return;
  }

  
const pointIsInsidePolygon = (latlng, points) => {
  const x = latlng.lng;
  const y = latlng.lat;
  let inside = false;

  for (
    let i = 0, j = points.length - 1;
    i < points.length;
    j = i++
  ) {
    const yi = points[i][0];
    const xi = points[i][1];
    const yj = points[j][0];
    const xj = points[j][1];

    const intersects =
      (yi > y) !== (yj > y) &&
      x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;

    if (intersects) {
      inside = !inside;
    }
  }

  return inside;
};

const clickedZone = Object.entries(zones)
  .reverse()
  .find(([id, zone]) => {
    return pointIsInsidePolygon(
      event.latlng,
      zone.points
    );
  });


  if (!clickedZone) {
    alert("Aucun quartier trouvé à cet endroit.");
    return;
  }

  const [id, zone] = clickedZone;

  selectedZoneId = id;
  drawing.points = zone.points.map(point => [...point]);

  const colors = window.D8Zones.colors;
  drawing.color = colors[zone.category] || "#bb65ff";

  refreshDrawing();

  selectingZone = false;
  zoneSelect.textContent = "📍 " + zone.name;
  map.getContainer().style.cursor = "";

  alert(
    "Quartier sélectionné : " + zone.name +
    "\nSes points sont maintenant affichés."
  );
});

    drawToggle.addEventListener("click", () => {
      drawing.enabled = !drawing.enabled;

      drawToggle.textContent = drawing.enabled
        ? "✓ Terminer le dessin"
        : "✏️ Activer le dessin";

      map.getContainer().style.cursor =
        drawing.enabled ? "crosshair" : "";

      drawTools.classList.toggle("drawing-active", drawing.enabled);
    });

    map.on("click", event => {
      if (!drawing.enabled) return;

      const lat = Math.round(event.latlng.lat);
      const lng = Math.round(event.latlng.lng);

      drawing.points.push([lat, lng]);
      refreshDrawing();
    });

    document.getElementById("d8-draw-undo")
      .addEventListener("click", () => {
        drawing.points.pop();
        refreshDrawing();
      });

    document.getElementById("d8-draw-clear")
      .addEventListener("click", () => {
        drawing.points = [];
        refreshDrawing();
      });

    document.getElementById("d8-draw-copy")
      .addEventListener("click", async () => {
        if (drawing.points.length < 3) {
          alert("Place au moins 3 points sur la carte.");
          return;
        }

        const coordinates =
          "points: " + JSON.stringify(drawing.points, null, 2);

        try {
          await navigator.clipboard.writeText(coordinates);
          alert("Coordonnées copiées !");
        } catch {
          window.prompt(
            "Copie ces coordonnées avec Ctrl + C :",
            coordinates
          );
        }
      });
  };
})();
